   115→function haversineKm(a: [number, number], b: [number, number]): number {
   116→  const dLat = (b[0] - a[0]) * Math.PI / 180;
   117→  const dLon = (b[1] - a[1]) * Math.PI / 180;
   118→  const lat1 = a[0] * Math.PI / 180;
   119→  const lat2 = b[0] * Math.PI / 180;
   120→  const value = Math.sin(dLat / 2) ** 2 + Math.sin(dLon / 2) ** 2 * Math.cos(lat1) * Math.cos(lat2);
   121→  return 6371 * 2 * Math.atan2(Math.sqrt(value), Math.sqrt(1 - value));
   122→}
   123→
   124→function nearestRouteIndex(point: [number, number], geometry: [number, number][]): number {
   125→  return geometry.reduce((nearestIndex, coordinate, index) =>
   126→    haversineKm(point, coordinate) < haversineKm(point, geometry[nearestIndex]) ? index : nearestIndex, 0);
   127→}

   489→  assessIncidentImpact: async (incidentId) => {

   492→    const incident = state.incidents.find(i => i.id === incidentId);

   498→
   499→    const affectedShipments = state.shipments.filter(s => {
   500→      if (s.status !== 'In Transit') return false;
   501→      const vehicle = state.vehicles.find(v => v.id === s.assignedVehicleId);
   502→      if (!vehicle || !vehicle.currentRouteId) return false;
   503→      if (vehicle.currentRouteGeometry?.length) {
   504→        const vehicleIndex = nearestRouteIndex(vehicle.coordinates, vehicle.currentRouteGeometry);
   505→        const incidentIndex = nearestRouteIndex(incident.coordinates, vehicle.currentRouteGeometry);
   506→        return incidentIndex > vehicleIndex &&
   507→          haversineKm(incident.coordinates, vehicle.currentRouteGeometry[incidentIndex]) <= 35;
   508→      }
   509→      const activeRoute = state.activeRoutes.find(r => r.id === vehicle.currentRouteId);
   510→      return Boolean(activeRoute?.corridorIds.includes(incident.affectedCorridorId ?? ''));
   511→    });
   512→
   513→    set(s => ({ incidents: s.incidents.map(i => i.id === incidentId ? {
   514→      ...i,
   515→      impactAssessment: {
   516→        assessedAt: timeNow,
   517→        affectedShipmentIds: affectedShipments.map(shipment => shipment.id),
   518→        affectsRemainingRoute: affectedShipments.length > 0,
   519→        recommendedAction: affectedShipments.length > 0 ? 'ROUTE_CHANGE' : 'MONITOR'
   520→      }
   521→    } : i) }));
   522→
   523→    if (affectedShipments.length > 0) {
   524→      get().recalculateNetwork();
   525→
   526→      for (const shipment of affectedShipments) {
   527→        const vehicle = state.vehicles.find(item => item.id === shipment.assignedVehicleId);
   528→        const destination = resolveLocationCoordinates(shipment.destination);
   529→        if (!vehicle || !destination) continue;
   530→        // If it's a hard blockage, pause the vehicle
   531→        const unsafe = (incident.type === 'Landslide' || incident.type === 'Road Blockage' || incident.type === 'Bridge Damage') && incident.severity === 'Critical';

   533→          set(s => ({
   534→            shipments: s.shipments.map(sh => sh.id === shipment.id ? { ...sh, status: 'Route Change Pending' } : sh),
   535→            vehicles: s.vehicles.map(v => v.id === shipment.assignedVehicleId ? { ...v, status: 'Paused for Safety' } : v)
   536→          }));
   537→          get().addAlert({
   538→            type: 'CRITICAL', title: 'Vehicle Paused for Safety',
   539→            message: `Vehicle ${shipment.assignedVehicleId} paused due to verified ${incident.type} ahead.`,
   540→            severity: 'High', recipientRole: 'Driver', actionRequired: false, actionTaken: false
   541→          });
   542→        }
   543→
   544→        // Generate alternatives
   545→        const alternatives = await generateRoutesAsync(shipment, get().incidents, state.simulationMode, vehicle.coordinates, destination);
   546→        const currentRouteAlt = state.activeRoutes.find(a => a.id === shipment.routeId);
   547→        const bestRoute = alternatives.find(a => a.isFeasible && a.id !== shipment.routeId);
   548→        
   549→        if (bestRoute) {
   550→          const rec: RouteRecommendation = {
   551→            id: 'REC-' + Math.random().toString(36).substring(7),
   552→            shipmentId: shipment.id,
   553→            incidentId,
   554→            recommendedRouteId: bestRoute.id,
   555→            alternativeRoutes: alternatives,
   556→            reason: `Avoids affected corridor. Reduces risk from ${currentRouteAlt?.risk || 100}% to ${bestRoute.risk}%.`,
   557→            status: 'ACTIVE',
   558→            createdAt: new Date().toISOString(),
   559→            lastEvaluatedAt: new Date().toISOString(),
   560→            trigger: 'Mid-Journey Verification',
   561→            explanation: `NORTHLINK detected a verified unresolved ${incident.type.toLowerCase()} ahead on the current route. The affected route has ${currentRouteAlt?.risk ?? 100}% predicted risk; ${bestRoute.name} is recommended at ${bestRoute.risk}% risk with an estimated ${Math.floor(bestRoute.currentEta / 60)}h ${bestRoute.currentEta % 60}m journey.`
   562→          };
   563→          set(s => ({ routeRecommendations: [rec, ...s.routeRecommendations], activeRoutes: alternatives }));
   564→          import('@/services/assistantService').then(({ assistantService }) => {
   565→            assistantService.askQuestion(`Why was ${bestRoute.name} recommended for ${shipment.id}?`).then(({ reply }) => {
   566→              set(s => ({ routeRecommendations: s.routeRecommendations.map(item => item.id === rec.id ? { ...item, explanation: reply } : item) }));
   567→            });
   568→          });

   570→            type: 'ROUTE UPDATE', title: 'Route Change Recommended',
