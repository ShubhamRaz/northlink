   133→function haversineKm(a: [number, number], b: [number, number]): number {
   134→  const dLat = (b[0] - a[0]) * Math.PI / 180;
   135→  const dLon = (b[1] - a[1]) * Math.PI / 180;
   136→  const lat1 = a[0] * Math.PI / 180;
   137→  const lat2 = b[0] * Math.PI / 180;
   138→  const value = Math.sin(dLat / 2) ** 2 + Math.sin(dLon / 2) ** 2 * Math.cos(lat1) * Math.cos(lat2);
   139→  return 6371 * 2 * Math.atan2(Math.sqrt(value), Math.sqrt(1 - value));
   140→}
   141→
   142→function nearestRouteIndex(point: [number, number], geometry: [number, number][]): number {
   143→  return geometry.reduce((nearestIndex, coordinate, index) =>
   144→    haversineKm(point, coordinate) < haversineKm(point, geometry[nearestIndex]) ? index : nearestIndex, 0);
   145→}

The affected route has ${currentRouteAlt?.risk ?? 100}% predicted risk; ${bestRoute.name} is recommended at ${bestRoute.risk}% risk with an estimated ${Math.floor(bestRoute.currentEta / 60)}h ${bestRoute.currentEta % 60}m journey.`
   583→          };
